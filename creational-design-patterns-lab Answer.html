Pattern 01

Singleton

20 POINTS
Scenario
The university Subject Offering System needs a centralized activity logger for the Coordinator. Every time the Coordinator creates a subject, creates an offering, assigns a faculty member, or clones a section, the action must be recorded in one shared logger instance.
Concept
Singleton ensures that only one CoordinatorLogger instance exists throughout the application.
Complete PHP Code — Correct Answer
PHP Complete Implementation
Faculty Answer 
<?php

class CoordinatorLogger
{
    private static ?CoordinatorLogger $instance = null;
    private array $logs = [];

    private function __construct()
    {
    }

    public static function getInstance(): CoordinatorLogger
    {
        // ⭐ CORRECT ANSWER
        if (self::$instance === null) {
            self::$instance = new CoordinatorLogger();
        }

        // ⭐ CORRECT ANSWER
        return self::$instance;
    }

    public function log(
        string $coordinatorId,
        string $action,
        string $details
    ): void {
        $this->logs[] = [
            "coordinator_id" => $coordinatorId,
            "action" => $action,
            "details" => $details,
            "datetime" => date("Y-m-d H:i:s")
        ];
    }

    public function getLogs(): array
    {
        return $this->logs;
    }
}


/*
|--------------------------------------------------------------------------
| TEST
|--------------------------------------------------------------------------
*/

$logger1 = CoordinatorLogger::getInstance();

$logger2 = CoordinatorLogger::getInstance();

$logger1->log(
    "COORD-001",
    "CREATE SUBJECT",
    "IT202 - Web Development"
);

echo $logger1 === $logger2
    ? "PASS: Same logger instance."
    : "FAIL: Different logger instances.";

echo "<pre>";
print_r($logger1->getLogs());
echo "</pre>";

interface Subject

    public function getDetails(): string;

Pattern 05

Prototype

20 POINTS
Scenario
The Coordinator has already created Section A for a subject. Instead of creating another offering from scratch, the Coordinator can clone the existing offering and modify only the values that need to change.
Concept
Prototype answers: "How can I create another similar object without building it from scratch?"
Complete PHP Code — Correct Answer
PHP Complete Implementation
Faculty Answer 
<?php

/*
|--------------------------------------------------------------------------
| PROTOTYPE
|--------------------------------------------------------------------------
*/

class SubjectOfferingPrototype
{
    public string $academicYear;

    public string $semester;

    public string $subjectCode;

    public string $subjectName;

    public string $subjectType;

    public string $section;

    public string $schedule;

    public string $room;

    public string $faculty;

    public int $capacity;


    public function __construct(
        string $academicYear,
        string $semester,
        string $subjectCode,
        string $subjectName,
        string $subjectType,
        string $section,
        string $schedule,
        string $room,
        string $faculty,
        int $capacity
    ) {

        $this->academicYear =
            $academicYear;

        $this->semester =
            $semester;

        $this->subjectCode =
            $subjectCode;

        $this->subjectName =
            $subjectName;

        $this->subjectType =
            $subjectType;

        $this->section =
            $section;

        $this->schedule =
            $schedule;

        $this->room =
            $room;

        $this->faculty =
            $faculty;

        $this->capacity =
            $capacity;
    }
}


/*
|--------------------------------------------------------------------------
| ORIGINAL SUBJECT OFFERING
|--------------------------------------------------------------------------
*/

$sectionA =
    new SubjectOfferingPrototype(

        "2026-2027",

        "1st Semester",

        "IT202",

        "Web Development",

        "Lecture",

        "BSIT-2A",

        "MWF 8:00 AM - 9:00 AM",

        "Room 301",

        "Juan Dela Cruz",

        40
    );


/*
|--------------------------------------------------------------------------
| CLONE
|--------------------------------------------------------------------------
*/

// ⭐ CORRECT ANSWER
$sectionB = clone $sectionA;


/*
|--------------------------------------------------------------------------
| MODIFY CLONE
|--------------------------------------------------------------------------
*/

// ⭐ CORRECT ANSWER
$sectionB->section =
    "BSIT-2B";


// ⭐ CORRECT ANSWER
$sectionB->schedule =
    "TTh 10:00 AM - 11:30 AM";


// ⭐ CORRECT ANSWER
$sectionB->room =
    "Room 302";


// ⭐ CORRECT ANSWER
$sectionB->faculty =
    "Maria Santos";


/*
|--------------------------------------------------------------------------
| TEST
|--------------------------------------------------------------------------
*/

echo "<h3>Original Section</h3>";

echo "<pre>";

print_r($sectionA);

echo "</pre>";


echo "<h3>Cloned Section</h3>";

echo "<pre>";

print_r($sectionB);

echo "</pre>";


echo $sectionA === $sectionB

    ? "FAIL: Same object."

    : "PASS: Clone is a separate object.";
